{%- comment -%} Turns [[note]] markers into numbered notes at build time, so they work with JS off. Below the margin breakpoint the number is a checkbox label that opens the note under the line; at wide widths the note floats into the margin. See components/sidenotes.css. Usage: {% include sidenotes.html html=content %} {%- endcomment -%} {%- assign sidenote_count = 0 -%} {%- assign sidenote_chunks = include.html | split: '[[' -%} {%- for chunk in sidenote_chunks -%} {%- if forloop.first -%} {{- chunk -}} {%- elsif chunk contains ']]' -%} {%- assign sidenote_count = sidenote_count | plus: 1 -%} {%- assign sidenote_text = chunk | split: ']]' | first -%} {{ sidenote_count }} {{ sidenote_text }} {{- chunk | remove_first: sidenote_text | remove_first: ']]' -}} {%- else -%} [[{{- chunk -}} {%- endif -%} {%- endfor -%}
